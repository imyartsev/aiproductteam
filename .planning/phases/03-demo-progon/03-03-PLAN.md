import os
from pathlib import Path

# Найти папку проекта
projects = list(Path('projects').iterdir()) if Path('projects').exists() else []
if not projects:
    print('FAIL: папка projects/ пуста')
    exit(1)

# Взять последнюю созданную папку
latest = max(projects, key=lambda p: p.stat().st_mtime)
print(f'Проект: {latest}')

checks = []
# Наличие ключевых файлов
for filename in ['main.py', 'requirements.txt', 'Dockerfile']:
    exists = (latest / filename).exists()
    checks.append((exists, f'{filename} существует'))

# Наличие тестов
tests = list(latest.rglob('test_*.py'))
checks.append((len(tests) > 0, f'тестовые файлы: {len(tests)} найдено'))

# Содержимое main.py
if (latest / 'main.py').exists():
    content = (latest / 'main.py').read_text(encoding='utf-8')
    checks.append(('FastAPI' in content, 'main.py содержит FastAPI'))
    checks.append(('/health' in content, 'main.py содержит /health'))

failed = [desc for ok, desc in checks if not ok]
passed = [desc for ok, desc in checks if ok]

for desc in passed:
    print(f'  [OK] {desc}')
for desc in failed:
    print(f'  [FAIL] {desc}')

if failed:
    exit(1)
else:
    print('PASS')
